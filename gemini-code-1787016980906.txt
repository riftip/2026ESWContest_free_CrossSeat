// ====================================================================
// [CAMERA FRAME - 2.0m(W) x 1.55m(H) x 0.9m(D) SOLID MODULAR SYSTEM]
// ====================================================================
// FRAME : Width = 2000 mm, Height = 1550 mm, Depth = +/-450 mm
// PROFILE: 28 x 28 mm | BED: 250 x 250 mm
// ====================================================================

$fn = 40;

mode = "assembly"; // ["assembly", "plate"]
plate_no = 13;     // [1:13]

FRAME_W = 2000;
FRAME_H = 1550;
FRAME_D = 450;

PROFILE = 28;
WALL    = 4.5;
BODY    = 200;
PIN_LEN = 30;
CLEAR   = 0.25;

DIAG = sqrt(FRAME_H * FRAME_H + FRAME_D * FRAME_D);

VERTICAL_N     = ceil(FRAME_H / BODY);
BEAM_N         = ceil(FRAME_W / BODY);
DIAG_N         = ceil(DIAG / BODY);

VERTICAL_PARTS = VERTICAL_N * 2;
BEAM_PARTS     = BEAM_N;
DIAG_PARTS     = DIAG_N * 4;
LONG_TOTAL     = VERTICAL_PARTS + BEAM_PARTS + DIAG_PARTS; // 총 62개

PARTS_PER_PLATE = 6;
LONG_PLATES     = ceil(LONG_TOTAL / PARTS_PER_PLATE); // 11 Plates

BEAM_START = VERTICAL_PARTS;
DIAG_START = BEAM_START + BEAM_PARTS;

// --- 유틸리티 ---
module orient_to(v) {
    L = norm(v);
    if (L > 0.0001) {
        pitch = acos(v[2] / L);
        yaw   = atan2(v[1], v[0]);
        rotate([0, 0, yaw])
            rotate([0, pitch, 0])
                children();
    } else {
        children();
    }
}

module socket_cut(dir, depth=35) {
    L = norm(dir);
    u = dir / L;
    orient_to(u)
        translate([0, 0, depth / 2])
            cube([PROFILE + CLEAR * 2, PROFILE + CLEAR * 2, depth + 0.1], center=true);
}

// --- 프로파일 부재 ---
module print_member(L, kind) {
    INNER    = PROFILE - (2 * WALL);
    PIN_SIZE = INNER - (2 * CLEAR);

    difference() {
        cube([PROFILE, PROFILE, L]);

        if (kind != "first") {
            translate([(PROFILE - INNER)/2, (PROFILE - INNER)/2, -0.1])
                cube([INNER, INNER, PIN_LEN + 0.1]);
        }

        translate([WALL, WALL, PIN_LEN])
            cube([INNER, INNER, max(0.1, L - (2 * PIN_LEN))]);
    }

    if (kind != "last") {
        translate([(PROFILE - PIN_SIZE)/2, (PROFILE - PIN_SIZE)/2, L])
            cube([PIN_SIZE, PIN_SIZE, PIN_LEN]);
    }
}

module get_long_part(index) {
    if (index < BEAM_START) {
        local = index % VERTICAL_N;
        L = min(BODY, FRAME_H - local * BODY);
        kind = (local == 0) ? "first" : ((local == VERTICAL_N - 1) ? "last" : "middle");
        print_member(L, kind);
    } else if (index < DIAG_START) {
        local = index - BEAM_START;
        L = min(BODY, FRAME_W - local * BODY);
        kind = (local == 0) ? "first" : ((local == BEAM_N - 1) ? "last" : "middle");
        print_member(L, kind);
    } else if (index < LONG_TOTAL) {
        local = (index - DIAG_START) % DIAG_N;
        L = min(BODY, DIAG - local * BODY);
        kind = (local == 0) ? "first" : ((local == DIAG_N - 1) ? "last" : "middle");
        print_member(L, kind);
    }
}

// --- 상단 허브 (살두께 보존) ---
module top_hub() {
    HUB = 95;
    difference() {
        cube([HUB, HUB, HUB], center=true);

        translate([HUB/2, 0, 0]) rotate([0, -90, 0]) socket_cut([0, 0, 1], 35);
        translate([-HUB/2, 0, 0]) rotate([0, 90, 0]) socket_cut([0, 0, 1], 35);
        translate([0, 0, -HUB/2]) socket_cut([0, 0, 1], 35);
        translate([0, 0, -HUB/2]) socket_cut([0, -FRAME_D, FRAME_H], 35);
        translate([0, 0, -HUB/2]) socket_cut([0, FRAME_D, FRAME_H], 35);
    }
}

// --- 바닥 발 받침대 ---
module bottom_joint(kind) {
    BASE_W = 85; BASE_L = 85; BASE_H = 45;
    difference() {
        union() {
            translate([-BASE_W/2, -BASE_L/2, 0])
                cube([BASE_W, BASE_L, BASE_H]);
            translate([-BASE_W/2 - 10, -BASE_L/2 - 10, 0])
                cube([BASE_W + 20, BASE_L + 20, 8]);
        }

        for (sx = [-1, 1], sy = [-1, 1]) {
            translate([sx * (BASE_W/2 + 5), sy * (BASE_L/2 + 5), -1])
                cylinder(d=5.5, h=12);
        }

        if (kind == "vertical") {
            translate([0, 0, BASE_H]) socket_cut([0, 0, -1], 35);
        } else if (kind == "front") {
            translate([0, 0, BASE_H]) socket_cut([0, FRAME_D, -FRAME_H], 35);
        } else if (kind == "back") {
            translate([0, 0, BASE_H]) socket_cut([0, -FRAME_D, -FRAME_H], 35);
        }
    }
}

// --- 카메라 마운트 ---
module rpi_camera_mount() {
    PW = 125; PH = 95; CLAMP_THICK = 6;
    difference() {
        union() {
            cube([PW, PROFILE + (CLAMP_THICK * 2), PROFILE + (CLAMP_THICK * 2)]);
            translate([0, PROFILE + CLAMP_THICK * 2, 0])
                cube([PW, 5, PH]);
        }

        translate([-1, CLAMP_THICK, CLAMP_THICK])
            cube([PW + 2, PROFILE + CLEAR*2, PROFILE + CLEAR*2]);

        for (ox = [0, 58], oz = [0, 49]) {
            translate([15 + ox, PROFILE + CLAMP_THICK * 2 + 10, 25 + oz])
                rotate([90, 0, 0]) cylinder(d=2.8, h=15);
        }

        translate([102, PROFILE + CLAMP_THICK * 2 + 10, 50])
            rotate([90, 0, 0]) cylinder(d=13, h=15);

        for (dx = [-10.5, 10.5], dz = [-6.25, 6.25]) {
            translate([102 + dx, PROFILE + CLAMP_THICK * 2 + 10, 50 + dz])
                rotate([90, 0, 0]) cylinder(d=2.2, h=15);
        }
    }
}

// --- 플레이트 출력 모듈 ---
module print_plate() {
    %color("gray", 0.15)
        translate([-125, -125, -2]) cube([250, 250, 2]);

    if (plate_no >= 1 && plate_no <= LONG_PLATES) {
        start_idx = (plate_no - 1) * PARTS_PER_PLATE;
        for (k = [0 : PARTS_PER_PLATE - 1]) {
            idx = start_idx + k;
            if (idx < LONG_TOTAL) {
                translate([-90 + k * 36, -BODY/2, 0])
                    get_long_part(idx);
            }
        }
    } else if (plate_no == 12) {
        translate([-55, -65, 0]) bottom_joint("vertical");
        translate([55, -65, 0])  bottom_joint("vertical");
        translate([-55, 0, 0])   bottom_joint("front");
        translate([55, 0, 0])    bottom_joint("front");
        translate([-55, 65, 0])  bottom_joint("back");
        translate([55, 65, 0])   bottom_joint("back");
    } else if (plate_no == 13) {
        translate([-55, 55, 95/2])  top_hub();
        translate([55, 55, 95/2])   top_hub();
        translate([-62.5, -105, 0]) rpi_camera_mount();
    }
}

// --- 전체 조립 모듈 ---
module beam_line(p1, p2) {
    v = p2 - p1;
    translate(p1)
        orient_to(v)
            translate([-PROFILE/2, -PROFILE/2, 0])
                cube([PROFILE, PROFILE, norm(v)]);
}

module assembly() {
    color("silver") beam_line([0, 0, FRAME_H], [FRAME_W, 0, FRAME_H]);

    for (x = [0, FRAME_W]) {
        color("silver") beam_line([x, 0, 0], [x, 0, FRAME_H]);
        color("orange") beam_line([x, FRAME_D, 0], [x, 0, FRAME_H]);
        color("orange") beam_line([x, -FRAME_D, 0], [x, 0, FRAME_H]);

        translate([x, 0, FRAME_H]) top_hub();
        translate([x, 0, 0]) bottom_joint("vertical");
        translate([x, FRAME_D, 0]) bottom_joint("front");
        translate([x, -FRAME_D, 0]) bottom_joint("back");
    }

    translate([FRAME_W/2 - 62.5, -PROFILE/2 - 6, FRAME_H - PROFILE/2 - 6])
        color("lime") rpi_camera_mount();

    %color("gray", 0.25)
        translate([-100, -FRAME_D - 50, -8])
            cube([FRAME_W + 200, (FRAME_D + 50) * 2, 8]);
}

if (mode == "assembly") {
    assembly();
} else if (mode == "plate") {
    print_plate();
}
